## 5. 핵심 알고리즘 구현

### 5.1 Vibration Sensor 배치 알고리즘

#### 5.1.1 기본 로직 (3x3 그리드)
```python
def calculate_vs_positions(width: float, depth: float, offset: float = 300) -> list:
    """
    9개의 Vibration Sensor를 3x3 그리드로 배치
    
    Args:
        width: 거실 가로 길이 (mm)
        depth: 거실 세로 길이 (mm)
        offset: 벽면에서의 오프셋 (mm)
    
    Returns:
        List of sensor positions: [{"id": 1, "x": 750, "y": 600, "type": "VS"}, ...]
    """
    # 실제 배치 가능 영역
    effective_width = width - (2 * offset)
    effective_depth = depth - (2 * offset)
    
    # 3x3 그리드 간격 계산
    x_spacing = effective_width / 2
    y_spacing = effective_depth / 2
    
    positions = []
    sensor_id = 1
    
    for row in range(3):
        for col in range(3):
            x = offset + (col * x_spacing)
            y = offset + (row * y_spacing)
            positions.append({
                "id": sensor_id,
                "x": round(x, 1),
                "y": round(y, 1),
                "type": "VS"
            })
            sensor_id += 1
    
    return positions
```

#### 5.1.2 간섭 회피 로직
```python
from scipy.spatial.distance import euclidean

def avoid_obstacles(positions: list, obstacles: list, min_distance: float = 300) -> list:
    """
    장애물과 센서 간의 최소 거리를 유지하도록 센서 위치 조정
    
    Args:
        positions: 센서 위치 리스트
        obstacles: 장애물 위치 리스트
        min_distance: 최소 안전 거리 (mm)
    
    Returns:
        조정된 센서 위치 리스트
    """
    adjusted_positions = []
    
    for pos in positions:
        sensor_point = np.array([pos["x"], pos["y"]])
        needs_adjustment = False
        
        # 각 장애물과의 거리 체크
        for obs in obstacles:
            obs_point = np.array([obs["x"], obs["y"]])
            distance = euclidean(sensor_point, obs_point)
            
            if distance < min_distance:
                needs_adjustment = True
                # 장애물 반대 방향으로 이동
                direction = sensor_point - obs_point
                direction = direction / np.linalg.norm(direction)  # 단위 벡터
                new_point = obs_point + direction * min_distance
                sensor_point = new_point
        
        adjusted_positions.append({
            **pos,
            "x": round(sensor_point[0], 1),
            "y": round(sensor_point[1], 1),
            "adjusted": needs_adjustment
        })
    
    return adjusted_positions
```

### 5.2 Speaker 배치 알고리즘

#### 5.2.1 Edge 기반 배치
```python
def calculate_spk_positions(width: float, depth: float, 
                           total_speakers: int = 12, 
                           offset: float = 100) -> list:
    """
    우물천장 라인을 따라 스피커 배치
    
    Args:
        width: 거실 가로 길이 (mm)
        depth: 거실 세로 길이 (mm)
        total_speakers: 총 스피커 개수
        offset: edge로부터의 오프셋 (mm)
    
    Returns:
        List of speaker positions
    """
    perimeter = 2 * (width + depth)
    spacing = perimeter / total_speakers
    
    positions = []
    speaker_id = 1
    current_length = 0
    
    edges = [
        ("top", width, 0, 1),      # 상단: 좌→우
        ("right", depth, 0, 1),    # 우측: 상→하
        ("bottom", width, 1, -1),  # 하단: 우→좌
        ("left", depth, 1, -1)     # 좌측: 하→상
    ]
    
    for edge_name, edge_length, start_max, direction in edges:
        num_speakers_on_edge = int(edge_length / spacing) + 1
        
        for i in range(num_speakers_on_edge):
            if speaker_id > total_speakers:
                break
                
            if edge_name == "top":
                x = i * spacing
                y = offset
            elif edge_name == "right":
                x = width - offset
                y = i * spacing
            elif edge_name == "bottom":
                x = width - (i * spacing)
                y = depth - offset
            else:  # left
                x = offset
                y = depth - (i * spacing)
            
            positions.append({
                "id": speaker_id,
                "x": round(x, 1),
                "y": round(y, 1),
                "type": "SPK",
                "edge": edge_name
            })
            speaker_id += 1
            
            if speaker_id > total_speakers:
                break
    
    return positions[:total_speakers]
```

### 5.3 케이블 길이 계산

#### 5.3.1 최단 경로 기반 계산 (Manhattan Distance)
```python
def calculate_cable_requirements(vs_positions: list, spk_positions: list, 
                                 controller_pos: dict) -> dict:
    """
    케이블 길이 및 규격별 수량 계산
    
    Args:
        vs_positions: VS 위치 리스트
        spk_positions: SPK 위치 리스트
        controller_pos: 제어기 위치
    
    Returns:
        케이블 요구사항 딕셔너리
    """
    cable_specs = {
        1.2: 0,  # 1.2m 케이블 개수
        7.0: 0   # 7.0m 케이블 개수
    }
    
    total_length = 0
    
    # 모든 디바이스 (VS + SPK)
    all_devices = vs_positions + spk_positions
    
    # 제어기로부터 각 디바이스까지의 맨해튼 거리 계산
    ctrl_point = np.array([controller_pos["x"], controller_pos["y"]])
    
    for device in all_devices:
        device_point = np.array([device["x"], device["y"]])
        # 맨해튼 거리 (벽면/천장을 따라 배선)
        distance = np.abs(device_point[0] - ctrl_point[0]) + \
                   np.abs(device_point[1] - ctrl_point[1])
        
        # mm를 m로 변환
        distance_m = distance / 1000
        total_length += distance_m
        
        # 적절한 케이블 규격 선택
        if distance_m <= 1.2:
            cable_specs[1.2] += 1
        elif distance_m <= 7.0:
            cable_specs[7.0] += 1
        else:
            # 7m 초과인 경우 7m 케이블 여러 개 사용
            cable_specs[7.0] += int(np.ceil(distance_m / 7.0))
    
    return {
        "total_length": round(total_length, 2),
        "cable_1_2m": cable_specs[1.2],
        "cable_7_0m": cable_specs[7.0]
    }
```

### 5.4 PDF 생성

#### 5.4.1 Shop 도면 생성
```python
from reportlab.lib.pagesizes import A3
from reportlab.pdfgen import canvas
from reportlab.lib.units import mm

def generate_shop_drawing_pdf(house_data: dict, design_data: dict, 
                               output_path: str) -> str:
    """
    Shop 도면 PDF 생성
    
    Args:
        house_data: 세대 정보
        design_data: 설계 데이터
        output_path: PDF 저장 경로
    
    Returns:
        생성된 PDF 파일 경로
    """
    c = canvas.Canvas(output_path, pagesize=A3)
    width, height = A3
    
    # 제목
    c.setFont("Helvetica-Bold", 20)
    c.drawString(30*mm, height - 30*mm, f"ANC Shop Drawing - {house_data['house_number']}")
    
    # 프로젝트 정보
    c.setFont("Helvetica", 10)
    info_y = height - 45*mm
    c.drawString(30*mm, info_y, f"Project: {house_data['project_name']}")
    c.drawString(30*mm, info_y - 5*mm, f"Living Room: {design_data['width']}mm x {design_data['depth']}mm")
    c.drawString(30*mm, info_y - 10*mm, f"Design Date: {design_data['design_date']}")
    
    # 도면 영역 설정 (scale 조정)
    drawing_origin_x = 50*mm
    drawing_origin_y = 100*mm
    scale = 0.05  # 1mm = 0.05mm on paper
    
    # 거실 외곽선 그리기
    room_width = design_data['width'] * scale
    room_depth = design_data['depth'] * scale
    c.rect(drawing_origin_x, drawing_origin_y, room_width, room_depth)
    
    # Vibration Sensor 그리기 (파란색 원)
    c.setFillColorRGB(0, 0, 1)
    for vs in design_data['vs_positions']:
        x = drawing_origin_x + (vs['x'] * scale)
        y = drawing_origin_y + (vs['y'] * scale)
        c.circle(x, y, 2*mm, fill=1)
        c.setFont("Helvetica", 6)
        c.drawString(x + 3*mm, y, f"VS-{vs['id']}")
    
    # Speaker 그리기 (빨간색 사각형)
    c.setFillColorRGB(1, 0, 0)
    for spk in design_data['spk_positions']:
        x = drawing_origin_x + (spk['x'] * scale)
        y = drawing_origin_y + (spk['y'] * scale)
        c.rect(x - 1.5*mm, y - 1.5*mm, 3*mm, 3*mm, fill=1)
        c.setFont("Helvetica", 6)
        c.drawString(x + 3*mm, y, f"SPK-{spk['id']}")
    
    # 범례
    legend_y = 50*mm
    c.setFillColorRGB(0, 0, 1)
    c.circle(30*mm, legend_y, 2*mm, fill=1)
    c.setFillColorRGB(0, 0, 0)
    c.drawString(35*mm, legend_y - 1*mm, "Vibration Sensor (VS)")
    
    c.setFillColorRGB(1, 0, 0)
    c.rect(28.5*mm, legend_y - 10*mm, 3*mm, 3*mm, fill=1)
    c.setFillColorRGB(0, 0, 0)
    c.drawString(35*mm, legend_y - 11*mm, "Speaker (SPK)")
    
    c.save()
    return output_path
```

    ssl_certificate_key /etc/nginx/ssl/key.pem;

    # Frontend
    location / {
        proxy_pass http://frontend;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }

    # Backend API
    location /api/ {
        proxy_pass http://backend;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # 파일 업로드 크기 제한
